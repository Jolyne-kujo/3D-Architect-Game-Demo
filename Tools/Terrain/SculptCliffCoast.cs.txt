// A small number of asymmetric, piecewise planar ridges describes uplifted beds.
// Each crest is hand placed; its two sides have different slope angles.
float Ridge(float x, float z, Vector3[] crest, float leftSlope, float rightSlope, float endSlope)
{
    float best = -100;
    for (int i = 0; i < crest.Length - 1; i++)
    {
        var a = crest[i]; var b = crest[i + 1];
        var delta = new Vector2(b.x - a.x, b.z - a.z);
        var rel = new Vector2(x - a.x, z - a.z);
        float along = Vector2.Dot(rel, delta) / delta.sqrMagnitude;
        float t = Mathf.Clamp01(along);
        float cross = (delta.x * rel.y - delta.y * rel.x) / delta.magnitude;
        float beyond = Mathf.Abs(along - t) * delta.magnitude;
        float flank = Mathf.Max(0, Mathf.Abs(cross) - 3.1f);
        float loss = (Mathf.Min(flank, 9) * .64f + Mathf.Max(0, flank - 9) * 1.2f) * (cross > 0 ? leftSlope : rightSlope);
        float y = Mathf.Lerp(a.y, b.y, t) - loss - beyond * endSlope;
        best = Mathf.Max(best, y);
    }
    return best;
}

bool Inside(float x,float z,Vector2[] polygon)
{
    bool inside=false;
    for(int i=0,j=polygon.Length-1;i<polygon.Length;j=i++)
    {var a=polygon[i];var b=polygon[j];if((a.y>z)!=(b.y>z)&&x<(b.x-a.x)*(z-a.y)/(b.y-a.y)+a.x)inside=!inside;}
    return inside;
}
void Road(float x, float z, out float distance, out float level, out Vector2 nearest)
{
    distance = 10000; level = 0;nearest=Vector2.zero;
    for (int i = 0; i < route.Length - 1; i++)
    {
        var a = route[i]; var b = route[i + 1];
        var delta = new Vector2(b.x - a.x, b.z - a.z);
        float t = Mathf.Clamp01(Vector2.Dot(new Vector2(x - a.x, z - a.z), delta) / delta.sqrMagnitude);
        float d = Vector2.Distance(new Vector2(x, z), new Vector2(a.x, a.z) + delta * t);
        if (d < distance) { distance = d; level = Mathf.Lerp(a.y, b.y, t);nearest=new Vector2(a.x,a.z)+delta*t; }
    }
}

float RockFacet(float x, float z)
{
    // Low-frequency triangular facets add a few large broken planes, not a noisy texture.
    float u = (x + z * .23f) / 11, v = (z - x * .17f) / 13;
    int ix = Mathf.FloorToInt(u), iz = Mathf.FloorToInt(v); u -= ix; v -= iz;
    float Knot(int a, int b) {float q = Mathf.Sin(a * 127.1f + b * 311.7f) * 43758.5453f; return (q - Mathf.Floor(q)) * 2 - 1;}
    float a = Knot(ix, iz), b = Knot(ix + 1, iz), c = Knot(ix, iz + 1), d = Knot(ix + 1, iz + 1);
    return u + v <= 1 ? a + (b - a) * u + (c - a) * v : d + (c - d) * (1 - u) + (b - d) * (1 - v);
}

{
    float wx = -130 + x * 300f / (resolution - 1), wz = -90 + z * 330f / (resolution - 1);
    float coast = Ease(-47, -9, wz);
    float lowCoastWidth = Mathf.Lerp(42, 80, Ease(8, 55, wz));
    float land = -6 + 6.3f * coast * (1 - Ease(lowCoastWidth, lowCoastWidth + 24, Mathf.Abs(wx + 8)));
    // Summit block bounded by inclined planes, not a round Gaussian hill.
    float summit = Mathf.Min(65 + (wx - 16) * .018f,
        .69f * (wz - 78) + .08f * (wx - 10),
        2.3f * (240 - wz) + .1f * wx,
        .92f * (wx + 80) + .10f * (wz - 140),
        1.45f * (110 - wx) - .10f * (wz - 155));
    float y = Mathf.Max(land, summit,
        Ridge(wx, wz, frontCrest, .96f, .64f, 1.05f),
        Ridge(wx, wz, westCrest, 1.35f, .63f, 1.0f),
        Ridge(wx, wz, eastCrest, .70f, 1.55f, 1.05f),
        Ridge(wx, wz, rearCrest, .65f, 1.65f, 1.1f));
    // Sparse erosion notches cut across the bedding; broad planes stay legible.
    float high = Ease(7, 28, y);
    float notch1 = Mathf.Max(0, 1 - Mathf.Abs(wx + .27f * wz + 25) / 5.5f);
    float notch2 = Mathf.Max(0, 1 - Mathf.Abs(wx - .18f * wz - 66) / 4.0f);
    y += high * (RockFacet(wx, wz) * 3.0f - notch1 * 3.4f - notch2 * 2.8f);

    Road(wx, wz, out float distance, out float roadLevel,out Vector2 nearest);
    bool onMesh=Inside(nearest.x,nearest.y,midFootprint)||Inside(nearest.x,nearest.y,summitFootprint);
    bool midGate=wx>41&&wx<49&&(Mathf.Abs(wz-91)<4||Mathf.Abs(wz-106)<4);
    bool peakGate=wx>-33&&wx<-25&&wz>146&&wz<159;
    float roadOnTerrain=(!onMesh||midGate||peakGate)?1:0;
    float terrace = TerraceWeight(wx, wz);
    // Wide earth shoulders support a genuinely cut-in road. They merge into the
    // mountain over 20 m instead of leaving a raised ribbon or retaining wall.
    float bank = 4.1f + 1.25f * Mathf.Sin(wz * .061f + .6f);
    float support = roadLevel + bank - distance * .16f - distance * distance * .027f;
    float shoulderWeight = Ease(18, 40, wz) * (1 - terrace)*roadOnTerrain;
    y = Mathf.Lerp(y, Mathf.Max(y, support), shoulderWeight);
    y = Mathf.Lerp(y, roadLevel - .04f, (1 - Ease(2.5f, 6.8f, distance))*roadOnTerrain);
    // Preserve the courtyard's foundation, its Terrain hole, and the sandy entry.
    float court = Mathf.Max(Mathf.Abs(wx) / 22, Mathf.Abs(wz) / 24);
    y = Mathf.Lerp(y, -.22f, 1 - Ease(1, 1.35f, court));
    float beach = (1 - Ease(32, 51, Mathf.Abs(wx + 25))) * (1 - Ease(19, 27, Mathf.Abs(wz + 46)));
    y = Mathf.Lerp(y, Mathf.Lerp(-3.7f, 0, Mathf.InverseLerp(-70, -27, wz)), beach);
    // Carve after the courtyard flattening so the last descent has no height seam.
    y = Mathf.Lerp(y, roadLevel - .04f, (1 - Ease(2.5f, 6.8f, distance))*roadOnTerrain);
    // Lower the ground below each solid cliff. Their tops/colliders carry the
    // walker; no heightfield surface fills or fakes the undercut faces.
    if(Inside(wx,wz,midFootprint)&&!midGate)y=Mathf.Min(y,5 + .13f*(wz-73));
    if(Inside(wx,wz,summitFootprint)&&!peakGate)y=Mathf.Min(y,43 + .2f*(wz-122));
    // A low notch frames the eastern bay from the halfway platform. It clears
    // the eye ray above the lower trail while the approach below remains occluded.
    float viewNotch=Ease(46,55,wx)*(1-Ease(85,99,wx))*(1-Ease(4,8,Mathf.Abs(wz-(82+(wx-41)*.082f))));
    y=Mathf.Lerp(y,Mathf.Min(y,25-.08f*(wx-60)),viewNotch);
    // Expose the front of the independent halfway block, leaving its right approach intact.
    float midApron=(1-Ease(25,34,Mathf.Abs(wx-17)))*(1-Ease(68,74,wz))*Ease(49,61,wz);
    y=Mathf.Lerp(y,Mathf.Min(y,6.5f),midApron);
    heights[z, x] = Mathf.Clamp01((y + 10) / 115);
}

UnityEngine.ProBuilder.ProBuilderMesh Cliff(string name,Vector2[] top,float topY,float lipY,float baseY,Vector2 bottomCenter,float sx,float sz)
{
    var vertices=new System.Collections.Generic.List<Vector3>();var faces=new System.Collections.Generic.List<UnityEngine.ProBuilder.Face>();
    Vector2 center=Vector2.zero;foreach(var p in top)center+=p;center/=top.Length;
    void Face(Vector3[] points,Vector3 outward,int material)
    {
        int start=vertices.Count;vertices.AddRange(points);var indexes=new System.Collections.Generic.List<int>();
        bool reverse=Vector3.Dot(Vector3.Cross(points[1]-points[0],points[2]-points[0]),outward)<0;
        for(int j=1;j<points.Length-1;j++){indexes.Add(start);indexes.Add(start+(reverse?j+1:j));indexes.Add(start+(reverse?j:j+1));}
        var f=new UnityEngine.ProBuilder.Face(indexes);f.submeshIndex=material;f.smoothingGroup=0;faces.Add(f);
    }
    var topRing=top.Select(p=>new Vector3(p.x,topY,p.y)).ToArray();
    var lipRing=top.Select(p=>new Vector3(Mathf.Lerp(center.x,p.x,.975f),lipY,Mathf.Lerp(center.y,p.y,.975f))).ToArray();
    var baseRing=top.Select(p=>new Vector3(bottomCenter.x+(p.x-center.x)*sx,baseY,bottomCenter.y+(p.y-center.y)*sz)).ToArray();
    Face(topRing,Vector3.up,0);Face(baseRing,Vector3.down,2);
    for(int i=0;i<top.Length;i++)
    {
        int j=(i+1)%top.Length;var side=(topRing[i]+topRing[j])*.5f-new Vector3(center.x,topY,center.y);
        Face(new[]{topRing[i],topRing[j],lipRing[j],lipRing[i]},side,1);
        Face(new[]{lipRing[i],lipRing[j],baseRing[j],baseRing[i]},side,2);
    }
    var existing=GameObject.Find(name);UnityEngine.ProBuilder.ProBuilderMesh pb;
    if(existing){pb=existing.GetComponent<UnityEngine.ProBuilder.ProBuilderMesh>();Undo.RecordObject(pb,"Reshape cliff block");pb.RebuildWithPositionsAndFaces(vertices,faces);}
    else{pb=UnityEngine.ProBuilder.ProBuilderMesh.Create(vertices,faces);pb.gameObject.name=name;Undo.RegisterCreatedObjectUndo(pb.gameObject,"Create cliff block");}
    pb.transform.SetParent(cliffRoot.transform,false);pb.transform.localPosition=Vector3.zero;
    pb.GetComponent<MeshRenderer>().sharedMaterials=new[]{capMat,faceMat,underMat};pb.ToMesh();pb.Refresh();
    var mesh=pb.GetComponent<MeshFilter>().sharedMesh;
    string meshPath="Assets/CoastalTemple/Terrain/"+name+".asset";
    if(!AssetDatabase.Contains(mesh))AssetDatabase.CreateAsset(mesh,meshPath);
    pb.preserveMeshAssetOnDestroy=true;
    var col=pb.GetComponent<MeshCollider>();if(!col)col=pb.gameObject.AddComponent<MeshCollider>();col.sharedMesh=null;col.sharedMesh=mesh;col.convex=false;
    EditorUtility.SetDirty(mesh);EditorUtility.SetDirty(pb);EditorUtility.SetDirty(col);
    return pb;
}
