{
    for(int i=0;i<segments;i++){var a=points[i];var b=points[i+1];var d=new Vector2(b.x-a.x,b.z-a.z);float t=Mathf.Clamp01(Vector2.Dot(new Vector2(x-a.x,z-a.z),d)/d.sqrMagnitude);float s=Vector2.Distance(new Vector2(x,z),new Vector2(a.x,a.z)+d*t);if(s<distance){distance=s;level=Mathf.Lerp(a.y,b.y,t);}}
}

float Carve(float x,float y,float z)
