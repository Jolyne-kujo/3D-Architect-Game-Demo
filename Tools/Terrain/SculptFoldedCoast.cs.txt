// Writes the existing native TerrainData; nothing generates terrain during Play.

Undo.RegisterCompleteObjectUndo(data, "Sculpt folded coastal mountain");
int resolution = data.heightmapResolution;
var heights = new float[resolution, resolution];
float Ease(float a, float b, float v) => Mathf.SmoothStep(0, 1, Mathf.InverseLerp(a, b, v));

// A small number of asymmetric, piecewise planar ridges describes uplifted beds.
// Each crest is hand placed; its two sides have different slope angles.
float Ridge(float x, float z, Vector3[] crest, float leftSlope, float rightSlope, float endSlope)
{
    float best = -100;
    for (int i = 0; i < crest.Length - 1; i++)
    {
        var a = crest[i]; var b = crest[i + 1];
        var delta = new Vector2(b.x - a.x, b.z - a.z);
        var rel = new Vector2(x - a.x, z - a.z);
        float along = Vector2.Dot(rel, delta) / delta.sqrMagnitude;
        float t = Mathf.Clamp01(along);
        float cross = (delta.x * rel.y - delta.y * rel.x) / delta.magnitude;
        float beyond = Mathf.Abs(along - t) * delta.magnitude;
        float flank = Mathf.Max(0, Mathf.Abs(cross) - 3.1f);
        float loss = (Mathf.Min(flank, 9) * .64f + Mathf.Max(0, flank - 9) * 1.2f) * (cross > 0 ? leftSlope : rightSlope);
        float y = Mathf.Lerp(a.y, b.y, t) - loss - beyond * endSlope;
        best = Mathf.Max(best, y);
    }
    return best;
}
var frontCrest = new[] {new Vector3(-45, 9, 43), new Vector3(-32, 20, 62), new Vector3(-9, 30, 88), new Vector3(19, 38, 114), new Vector3(43, 35, 131)};
var westCrest = new[] {new Vector3(-65, 8, 67), new Vector3(-63, 23, 102), new Vector3(-55, 37, 143), new Vector3(-65, 30, 168), new Vector3(-53, 18, 203)};
var eastCrest = new[] {new Vector3(37, 23, 64), new Vector3(52, 33, 82), new Vector3(67, 35, 105), new Vector3(87, 27, 127), new Vector3(97, 9, 151)};
var rearCrest = new[] {new Vector3(58, 35, 146), new Vector3(64, 43, 169), new Vector3(56, 39, 198), new Vector3(40, 18, 221)};
var route = new[] {
    new Vector3(-30,0,-27),new Vector3(-17,0,-15),new Vector3(-17,0,15),
    new Vector3(-28,3,32),new Vector3(-13,9,50),new Vector3(23,17,66),
    new Vector3(41,24,83),new Vector3(67,24,96),new Vector3(42,24,98),
    new Vector3(40,30,107),new Vector3(13,36,118),new Vector3(-8,40,137),
    new Vector3(16,46,140),new Vector3(16,48,148)
};
void Road(float x, float z, out float distance, out float level)
{
    distance = 10000; level = 0;
    for (int i = 0; i < route.Length - 1; i++)
    {
        var a = route[i]; var b = route[i + 1];
        var delta = new Vector2(b.x - a.x, b.z - a.z);
        float t = Mathf.Clamp01(Vector2.Dot(new Vector2(x - a.x, z - a.z), delta) / delta.sqrMagnitude);
        float d = Vector2.Distance(new Vector2(x, z), new Vector2(a.x, a.z) + delta * t);
        if (d < distance) { distance = d; level = Mathf.Lerp(a.y, b.y, t); }
    }
}
float TerraceWeight(float x, float z)
{
    // A slightly skewed polygonal bench, with an open seaward edge.
    float dx = x - 61, dz = z - 96;
    float u = dx * .98f + dz * .2f, v = dz * .98f - dx * .2f;
    float edge = Mathf.Max(Mathf.Abs(u) / 20, Mathf.Abs(v) / 11, (Mathf.Abs(u) + Mathf.Abs(v)) / 27);
    return 1 - Ease(.83f, 1.18f, edge);
}
float RockFacet(float x, float z)
{
    // Low-frequency triangular facets add a few large broken planes, not a noisy texture.
    float u = (x + z * .23f) / 11, v = (z - x * .17f) / 13;
    int ix = Mathf.FloorToInt(u), iz = Mathf.FloorToInt(v); u -= ix; v -= iz;
    float Knot(int a, int b) {float q = Mathf.Sin(a * 127.1f + b * 311.7f) * 43758.5453f; return (q - Mathf.Floor(q)) * 2 - 1;}
    float a = Knot(ix, iz), b = Knot(ix + 1, iz), c = Knot(ix, iz + 1), d = Knot(ix + 1, iz + 1);
    return u + v <= 1 ? a + (b - a) * u + (c - a) * v : d + (c - d) * (1 - u) + (b - d) * (1 - v);
}
for (int z = 0; z < resolution; z++) for (int x = 0; x < resolution; x++)
{
    float wx = -130 + x * 300f / (resolution - 1), wz = -90 + z * 330f / (resolution - 1);
    float coast = Ease(-47, -9, wz);
    float lowCoastWidth = Mathf.Lerp(42, 80, Ease(8, 55, wz));
    float land = -6 + 6.3f * coast * (1 - Ease(lowCoastWidth, lowCoastWidth + 24, Mathf.Abs(wx + 8)));
    // Summit block bounded by inclined planes, not a round Gaussian hill.
    float summit = Mathf.Min(48.6f + (wx - 16) * .018f,
        .91f * (wz - 91) + .08f * (wx - 10),
        2.0f * (222 - wz) + .1f * wx,
        .79f * (wx + 68) + .10f * (wz - 140),
        1.18f * (86 - wx) - .10f * (wz - 155));
    float y = Mathf.Max(land, summit,
        Ridge(wx, wz, frontCrest, .96f, .64f, 1.05f),
        Ridge(wx, wz, westCrest, 1.35f, .63f, 1.0f),
        Ridge(wx, wz, eastCrest, .70f, 1.55f, 1.05f),
        Ridge(wx, wz, rearCrest, .65f, 1.65f, 1.1f));
    // Sparse erosion notches cut across the bedding; broad planes stay legible.
    float high = Ease(7, 28, y);
    float notch1 = Mathf.Max(0, 1 - Mathf.Abs(wx + .27f * wz + 25) / 5.5f);
    float notch2 = Mathf.Max(0, 1 - Mathf.Abs(wx - .18f * wz - 66) / 4.0f);
    y += high * (RockFacet(wx, wz) * 3.0f - notch1 * 3.4f - notch2 * 2.8f);

    Road(wx, wz, out float distance, out float roadLevel);
    float terrace = TerraceWeight(wx, wz);
    // Wide earth shoulders support a genuinely cut-in road. They merge into the
    // mountain over 20 m instead of leaving a raised ribbon or retaining wall.
    float bank = 4.1f + 1.25f * Mathf.Sin(wz * .061f + .6f);
    float support = roadLevel + bank - distance * .16f - distance * distance * .027f;
    float shoulderWeight = Ease(18, 40, wz) * (1 - Ease(136, 148, wz)) * (1 - terrace);
    y = Mathf.Lerp(y, Mathf.Max(y, support), shoulderWeight);
    y = Mathf.Lerp(y, 23.94f, terrace);
    y = Mathf.Lerp(y, roadLevel - .04f, 1 - Ease(2.5f, 6.8f, distance));

    // Keep the authored temple foundation and approach clear and level.
    float summitPatch = Mathf.Max(Mathf.Abs(wx - 16) / 20.5f, Mathf.Abs(wz - 173) / 24.5f);
    y = Mathf.Lerp(y, 47.94f, 1 - Ease(1, 1.12f, summitPatch));
    // Preserve the courtyard's foundation, its Terrain hole, and the sandy entry.
    float court = Mathf.Max(Mathf.Abs(wx) / 22, Mathf.Abs(wz) / 24);
    y = Mathf.Lerp(y, -.22f, 1 - Ease(1, 1.35f, court));
    float beach = (1 - Ease(32, 51, Mathf.Abs(wx + 25))) * (1 - Ease(19, 27, Mathf.Abs(wz + 46)));
    y = Mathf.Lerp(y, Mathf.Lerp(-3.7f, 0, Mathf.InverseLerp(-70, -27, wz)), beach);
    // Carve after the courtyard flattening so the last descent has no height seam.
    y = Mathf.Lerp(y, roadLevel - .04f, 1 - Ease(2.5f, 6.8f, distance));
    heights[z, x] = Mathf.Clamp01((y + 10) / 80);
}
data.SetHeights(0, 0, heights);

TerrainLayer SolidLayer(string name, Color color)
{
    string root = "Assets/CoastalTemple/Terrain/";
    var texture = AssetDatabase.LoadAssetAtPath<Texture2D>(root + name + ".asset");
    if (!texture) { texture = new Texture2D(4, 4); texture.name = name; AssetDatabase.CreateAsset(texture, root + name + ".asset"); }
    texture.SetPixels(Enumerable.Repeat(color, 16).ToArray()); texture.Apply(); EditorUtility.SetDirty(texture);
    var layer = AssetDatabase.LoadAssetAtPath<TerrainLayer>(root + name + ".terrainlayer");
    if (!layer) {layer = new TerrainLayer(); AssetDatabase.CreateAsset(layer, root + name + ".terrainlayer");}
    layer.name = name; layer.diffuseTexture = texture; layer.tileSize = Vector2.one * 12;
    layer.smoothness = .06f; EditorUtility.SetDirty(layer); return layer;
}

var alpha = new float[ar, ar, 4];
for (int z = 0; z < ar; z++) for (int x = 0; x < ar; x++)
{
    float u = x / (float)(ar - 1), v = z / (float)(ar - 1);
    float wx = -130 + u * 300, wz = -90 + v * 330;
    float h = data.GetInterpolatedHeight(u, v) - 10;
    float slope = data.GetSteepness(u, v);
    Road(wx, wz, out float distance, out _);
    float sw = (1 - Ease(-31, -24, wz)) * (1 - Ease(32, 51, Mathf.Abs(wx + 25)));
    float pw = (1 - Ease(1.45f, 2.7f, distance)) * .52f * (1 - sw);
    float green = (1 - Ease(22, 43, slope)) * Ease(1.0f, 7, h) * (1 - Ease(43, 48, h));
    green *= .75f + .25f * Mathf.Sin(wx * .038f + wz * .026f);
    green *= (1 - sw) * (1 - pw);
    alpha[z, x, 0] = 1 - sw - pw - green; alpha[z, x, 1] = sw;
    alpha[z, x, 2] = pw; alpha[z, x, 3] = green;
}
data.SetAlphamaps(0, 0, alpha);
