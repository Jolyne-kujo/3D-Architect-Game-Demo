Mesh SavedMesh(string name,Mesh source,Vector3[] vertices,bool grassy=true)
{
    string path=output+name+".asset";var mesh=AssetDatabase.LoadAssetAtPath<Mesh>(path);
    if(!mesh){mesh=UnityEngine.Object.Instantiate(source);mesh.name=name;AssetDatabase.CreateAsset(mesh,path);}else EditorUtility.CopySerialized(source,mesh);
    mesh.name=name;mesh.vertices=vertices;mesh.RecalculateNormals();
    // Bake material regions and planar grass UVs once, without changing the
    // imported cliff's triangle geometry or collision surface.
    var normals=mesh.normals;var triangles=source.triangles;
    var points=vertices.ToList();var uv=source.uv.ToList();var newNormals=normals.ToList();
    var rockTriangles=new System.Collections.Generic.List<int>();var grassTriangles=new System.Collections.Generic.List<int>();
    var grassIndices=new System.Collections.Generic.Dictionary<int,int>();
    int GrassVertex(int old)
    {
        if(grassIndices.TryGetValue(old,out int index))return index;
        index=points.Count;grassIndices.Add(old,index);points.Add(vertices[old]);newNormals.Add(normals[old]);uv.Add(new Vector2(vertices[old].x,vertices[old].z)*.12f);return index;
    }
    for(int i=0;i<triangles.Length;i+=3)
    {
        int a=triangles[i],b=triangles[i+1],c=triangles[i+2];var normal=Vector3.Cross(vertices[b]-vertices[a],vertices[c]-vertices[a]).normalized;
        if(grassy&&normal.y>.72f){grassTriangles.Add(GrassVertex(a));grassTriangles.Add(GrassVertex(b));grassTriangles.Add(GrassVertex(c));}
        else{rockTriangles.Add(a);rockTriangles.Add(b);rockTriangles.Add(c);}
    }
    mesh.Clear();mesh.indexFormat=points.Count>65535?UnityEngine.Rendering.IndexFormat.UInt32:UnityEngine.Rendering.IndexFormat.UInt16;
    mesh.SetVertices(points);mesh.SetNormals(newNormals);mesh.SetUVs(0,uv);mesh.subMeshCount=grassTriangles.Count>0?2:1;
    mesh.SetTriangles(rockTriangles,0);if(grassTriangles.Count>0)mesh.SetTriangles(grassTriangles,1);
    mesh.RecalculateTangents();mesh.RecalculateBounds();EditorUtility.SetDirty(mesh);return mesh;
}
float SummitHeight(float x,float z)
{
    float h=22+(x+38)*(8f/103)+(z-51)*((16-110*(8f/103))/97);
    float pad=Mathf.Max(Mathf.Abs(x-40)/12.0f,Mathf.Abs(z-88)/15.0f);
    return Mathf.Lerp(h,31,1-Ease(1,1.3f,pad));
}
